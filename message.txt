const canvas = document.getElementById('game-canvas');
const ctx = canvas.getContext('2d');

// UI Elements
const uiStart = document.getElementById('ui-start');
const uiGameOver = document.getElementById('ui-game-over');
const btnStart = document.getElementById('btn-start');
const btnRestart = document.getElementById('btn-restart');
const uiScore = document.getElementById('ui-score');
const finalScore = document.getElementById('final-score');
const uiBossTimer = document.getElementById('ui-boss-timer');
const bossTime = document.getElementById('boss-time');

// Game Constants
const GRAVITY = 0.05;
const JUMP_POWER = -3.5;
const PLATFORM_WIDTH = 60;
const PLATFORM_HEIGHT = 10;
const PLAYER_SIZE = 20;
const CANVAS_WIDTH = 400;
const CANVAS_HEIGHT = 700;

// Game State
let animationId;
let platforms = [];
let player = {};
let score = 0;
let cameraY = 0;
let isPlaying = false;
let isBossFight = false;
let bossTimerFrames = 0;
let bossLevel = 1;
let nextBossScore = 1000;
let projectiles = [];
let boss = null;
let keys = {
    left: false,
    right: false
};

// Colors
const COLOR_PLAYER = '#ffffff';
const COLOR_PLATFORM_NORMAL = '#00f0ff';
const COLOR_PLATFORM_MOVING = '#ff003c';
const COLOR_BOSS = '#a200ff';
const COLOR_PROJECTILE = '#00ff00';
const COLOR_PARRY = '#ff00ff'; // Neon pink

class Boss {
    constructor(level) {
        this.width = 40;
        this.height = 40;
        this.x = CANVAS_WIDTH / 2 - this.width / 2;
        this.y = cameraY + 80;
        this.vx = 2 + (level * 0.5); // Scales with level
        this.fireRate = Math.max(30, 90 - (level * 2)); // Frames between shots
        this.frameCounter = 0;
    }

    update() {
        this.y = cameraY + 80; // Lock to camera so it stays at the top

        this.x += this.vx;
        if (this.x < 0 || this.x + this.width > CANVAS_WIDTH) {
            this.vx *= -1;
        }

        this.frameCounter++;
        if (this.frameCounter >= this.fireRate) {
            this.frameCounter = 0;
            this.shoot();
        }
    }

    shoot() {
        projectiles.push(new Projectile(this.x + this.width / 2, this.y + this.height));
    }

    draw() {
        ctx.fillStyle = COLOR_BOSS;
        ctx.shadowBlur = 15;
        ctx.shadowColor = COLOR_BOSS;
        ctx.fillRect(this.x, this.y - cameraY, this.width, this.height);

        // Angry eye glow
        ctx.fillStyle = '#ff0000';
        ctx.fillRect(this.x + 8, this.y - cameraY + 10, 6, 6);
        ctx.fillRect(this.x + 26, this.y - cameraY + 10, 6, 6);
        ctx.shadowBlur = 0;
    }
}

class Projectile {
    constructor(x, y) {
        this.x = x;
        this.y = y;
        this.radius = 8;
        this.vy = 4 + (bossLevel * 0.5);
        this.isPink = Math.random() < 0.9; // 30% chance for parryable pink cube
    }

    update() {
        this.y += this.vy;
    }

    draw() {
        ctx.fillStyle = this.isPink ? COLOR_PARRY : COLOR_PROJECTILE;
        ctx.shadowBlur = 10;
        ctx.shadowColor = ctx.fillStyle;
        ctx.beginPath();
        if (this.isPink) {
            let size = this.radius * 2;
            ctx.rect(this.x - this.radius, this.y - cameraY - this.radius, size, size);
        } else {
            ctx.arc(this.x, this.y - cameraY, this.radius, 0, Math.PI * 2);
        }
        ctx.fill();
        ctx.shadowBlur = 0;
    }
}

class Player {
    constructor() {
        this.x = CANVAS_WIDTH / 2 - PLAYER_SIZE / 2;
        this.y = CANVAS_HEIGHT / 2;
        this.width = PLAYER_SIZE;
        this.height = PLAYER_SIZE;
        this.vx = 0;
        this.vy = 0;
        this.speed = 3;
        this.parryFrames = 0;
    }

    update() {
        if (this.parryFrames > 0) {
            this.parryFrames--;
        }

        // Horizontal Movement
        if (keys.left) {
            this.vx = -this.speed;
        } else if (keys.right) {
            this.vx = this.speed;
        } else {
            this.vx = 0;
        }

        this.x += this.vx;

        // Apply Gravity
        this.vy += GRAVITY;
        this.y += this.vy;

        // Screen Wrap
        if (this.x + this.width < 0) {
            this.x = CANVAS_WIDTH;
        } else if (this.x > CANVAS_WIDTH) {
            this.x = -this.width;
        }
    }

    draw() {
        let isParrying = this.parryFrames > 0;
        ctx.fillStyle = isParrying ? COLOR_PARRY : COLOR_PLAYER;
        ctx.shadowBlur = isParrying ? 20 : 10;
        ctx.shadowColor = isParrying ? COLOR_PARRY : COLOR_PLAYER;
        ctx.fillRect(this.x, this.y - cameraY, this.width, this.height);

        // Glow effect
        ctx.fillStyle = 'rgba(255, 255, 255, 0.5)';
        ctx.fillRect(this.x + 2, this.y - cameraY + 2, this.width - 4, this.height - 4);

        ctx.shadowBlur = 0; // Reset shadow
    }

    jump(multiplier = 1) {
        this.vy = JUMP_POWER * multiplier;
    }
}

class Platform {
    constructor(x, y, type = 0) {
        this.x = x;
        this.y = y;
        this.width = PLATFORM_WIDTH;
        this.height = PLATFORM_HEIGHT;
        this.type = type; // 0: Normal, 1: Moving
        this.vx = (Math.random() > 0.5 ? 1 : -1) * (1 + Math.random());
    }

    update() {
        if (this.type === 1) {
            this.x += this.vx;
            if (this.x < 0 || this.x + this.width > CANVAS_WIDTH) {
                this.vx *= -1;
            }
        }
    }

    draw() {
        ctx.fillStyle = this.type === 0 ? COLOR_PLATFORM_NORMAL : COLOR_PLATFORM_MOVING;
        ctx.shadowBlur = 10;
        ctx.shadowColor = ctx.fillStyle;

        // Draw rounded rectangle for platform
        ctx.beginPath();
        ctx.roundRect(this.x, this.y - cameraY, this.width, this.height, 5);
        ctx.fill();
        ctx.shadowBlur = 0;
    }
}

function initGame() {
    player = new Player();
    platforms = [];
    score = 0;
    cameraY = 0;
    uiScore.innerText = score;

    projectiles = [];
    boss = null;
    isBossFight = false;
    bossLevel = 1;
    nextBossScore = 1000;
    uiBossTimer.classList.add('hidden');

    // Create initial platform under player
    platforms.push(new Platform(CANVAS_WIDTH / 2 - PLATFORM_WIDTH / 2, CANVAS_HEIGHT - 50));

    // Generate max 15 platforms spreading upwards
    for (let i = 1; i < 15; i++) {
        spawnPlatform(CANVAS_HEIGHT - 50 - i * 50);
    }
}

function spawnPlatform(yLevel) {
    let x = Math.random() * (CANVAS_WIDTH - PLATFORM_WIDTH);
    // 20% chance for a moving platform, increasing slightly with score but capped
    let isMoving = Math.random() < Math.min(0.2 + (score / 10000), 0.5) ? 1 : 0;
    platforms.push(new Platform(x, yLevel, isMoving));
}

function updateGame() {
    if (!isPlaying) return;

    player.update();

    // Trigger Boss Fight
    if (score >= nextBossScore && !isBossFight) {
        isBossFight = true;
        bossTimerFrames = 1200; // 20 seconds at 60fps
        boss = new Boss(bossLevel);
        projectiles = [];
        uiBossTimer.classList.remove('hidden');
    }

    if (isBossFight) {
        boss.update();

        // Timer management
        bossTimerFrames--;
        bossTime.innerText = Math.ceil(bossTimerFrames / 60);

        if (bossTimerFrames <= 0) {
            isBossFight = false;
            boss = null;
            projectiles = [];
            uiBossTimer.classList.add('hidden');
            bossLevel++;
            nextBossScore += 1000;
        }
    }

    // Collision Detection
    // Check collisions only when falling
    if (player.vy > 0) {
        for (let i = 0; i < platforms.length; i++) {
            let p = platforms[i];
            if (
                player.x < p.x + p.width &&
                player.x + player.width > p.x &&
                player.y + player.height > p.y &&
                player.y + player.height < p.y + p.height + player.vy // Ensure we hit the top and don't phase through
            ) {
                player.jump();
                break;
            }
        }
    }

    // Camera Logic
    // If player moves above the middle of the screen
    if (player.y < cameraY + CANVAS_HEIGHT / 2) {
        let diff = (cameraY + CANVAS_HEIGHT / 2) - player.y;
        cameraY -= diff;
        score += Math.floor(diff);
        uiScore.innerText = score;
    }

    // Platform Management
    platforms.forEach(p => p.update());

    // Remove platforms that are below the screen
    platforms = platforms.filter(p => p.y - cameraY < CANVAS_HEIGHT + 50);

    // Add new platforms above
    while (platforms[platforms.length - 1].y > cameraY - 50) {
        spawnPlatform(platforms[platforms.length - 1].y - (40 + Math.random() * 40));
    }

    // Handle Projectiles
    projectiles.forEach((p, index) => {
        p.update();

        // Circular collision with player
        let pCenterX = player.x + player.width / 2;
        let pCenterY = player.y + player.height / 2;
        let dist = Math.hypot(pCenterX - p.x, pCenterY - p.y);

        if (dist < p.radius + player.width / 2) {
            if (p.isPink && player.parryFrames > 0) {
                // Parry!
                player.jump(1.5); // Super jump!
                score += 500;
                uiScore.innerText = score;
                projectiles.splice(index, 1);
                player.parryFrames = 0; // Prevent double parry
            } else {
                gameOver();
            }
        }

        // Remove if off screen
        if (p.y - cameraY > CANVAS_HEIGHT) {
            projectiles.splice(index, 1);
        }
    });

    // Game Over condition
    if (player.y > cameraY + CANVAS_HEIGHT) {
        gameOver();
    }
}

function drawGame() {
    // Clear Canvas
    ctx.clearRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

    if (!isPlaying) return;

    // Draw grid background for that neon feel
    ctx.strokeStyle = 'rgba(0, 240, 255, 0.05)';
    ctx.lineWidth = 1;
    let gridOffset = cameraY % 50;

    for (let i = 0; i < CANVAS_WIDTH; i += 50) {
        ctx.beginPath();
        ctx.moveTo(i, 0);
        ctx.lineTo(i, CANVAS_HEIGHT);
        ctx.stroke();
    }

    for (let i = 0; i < CANVAS_HEIGHT; i += 50) {
        ctx.beginPath();
        ctx.moveTo(0, i - gridOffset);
        ctx.lineTo(CANVAS_WIDTH, i - gridOffset);
        ctx.stroke();
    }

    // Draw Entities
    platforms.forEach(p => p.draw());
    player.draw();
    if (isBossFight && boss) {
        boss.draw();
    }
    projectiles.forEach(p => p.draw());
}

function gameLoop() {
    updateGame();
    drawGame();
    if (isPlaying) {
        animationId = requestAnimationFrame(gameLoop);
    }
}

function startGame() {
    uiStart.classList.remove('active');
    uiStart.classList.add('hidden');
    uiGameOver.classList.remove('active');
    uiGameOver.classList.add('hidden');

    initGame();
    isPlaying = true;
    gameLoop();
}

function gameOver() {
    isPlaying = false;
    cancelAnimationFrame(animationId);

    finalScore.innerText = score;
    uiGameOver.classList.remove('hidden');
    uiGameOver.classList.add('active');
}

// Input Listeners
window.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') {
        keys.left = true;
    }
    if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') {
        keys.right = true;
    }
});

window.addEventListener('keyup', (e) => {
    if (e.key === 'ArrowLeft' || e.key === 'a' || e.key === 'A') {
        keys.left = false;
    }
    if (e.key === 'ArrowRight' || e.key === 'd' || e.key === 'D') {
        keys.right = false;
    }
});

btnStart.addEventListener('click', startGame);
btnRestart.addEventListener('click', startGame);

window.addEventListener('mousedown', (e) => {
    // Only apply parry if playing and not clicking on the buttons
    if (isPlaying && e.target.tagName !== 'BUTTON') {
        player.parryFrames = 15; // 0.25 seconds parry window at 60fps
    }
});

// Draw an initial frame for the background while waiting
ctx.fillStyle = '#0f172a';
ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);
